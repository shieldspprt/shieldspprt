Use AsyncLocalStorage to propagate request context through async call chains without threading parameters manually

```typescript
import { AsyncLocalStorage } from "async_hooks";

type RequestContext = {
  traceId: string;
  userId: string;
  requestId: string;
};

const contextStorage = new AsyncLocalStorage<RequestContext>();

function getContext(): RequestContext | undefined {
  return contextStorage.getStore();
}

function withContext<T>(ctx: RequestContext, fn: () => T): T {
  return contextStorage.run(ctx, fn);
}

// Middleware: extract headers and run the handler with context
app.use((req, res, next) => {
  const ctx: RequestContext = {
    traceId: req.headers["x-trace-id"] as string || crypto.randomUUID(),
    userId: req.headers["x-user-id"] as string || "anonymous",
    requestId: req.headers["x-request-id"] as string || crypto.randomUUID(),
  };
  withContext(ctx, next);
});

// Anywhere in the call chain (tool calls, DB queries, LLM calls):
function logToolCall(tool: string, args: unknown) {
  const ctx = getContext();
  console.log(JSON.stringify({ ...ctx, tool, args, ts: Date.now() }));
}
```

- `AsyncLocalStorage` preserves context across `await` boundaries, promises, and callbacks
- Zero parameter threading: downstream code calls `getContext()` instead of receiving context as an argument
- Essential for observability in agent systems where a single request fans out to multiple tool calls and LLM providers
