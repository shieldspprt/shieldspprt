Eval-driven prompt development beats manual prompt engineering by converting subjective tuning into measurable regression tests. Define a small eval set with expected behaviors, score each prompt change against it, and only merge when the aggregate passes your threshold.

```typescript
interface EvalCase {
  input: string
  expected: {
    toolCalls?: string[]
    mustContain?: string[]
    mustNotContain?: string[]
    schema?: object
  }
}

async function runEval(
  prompt: string,
  cases: EvalCase[],
  runAgent: (prompt: string, input: string) => Promise<AgentResult>
): Promise<EvalReport> {
  const results = await Promise.all(
    cases.map(async (c) => {
      const result = await runAgent(prompt, c.input)
      return { case: c, passed: scoreResult(result, c) }
    })
  )
  const passRate = results.filter((r) => r.passed).length / cases.length
  return { passRate, details: results }
}

function scoreResult(result: AgentResult, expected: EvalCase["expected"]): boolean {
  if (expected.toolCalls && !expected.toolCalls.every((t) => result.tools.includes(t))) return false
  if (expected.mustContain && !expected.mustContain.every((s) => result.text.includes(s))) return false
  if (expected.mustNotContain && expected.mustNotContain.some((s) => result.text.includes(s))) return false
  if (expected.schema && !validateSchema(result.structured, expected.schema)) return false
  return true
}
```

- Treat prompts like code: version them, test them, and gate merges on eval pass rate.
- Keep eval sets small (20 to 50 cases) and focused on regression risk, not coverage theater.
- Run evals in CI on every prompt change so a regression blocks deploy automatically.
