# Propagate a request ID through every async hop so logs correlate end to end

When an LLM agent fans out tool calls across services a single user request can spawn dozens of log lines. A request ID generated at the edge and passed through every async boundary lets you reconstruct the full trace in one grep.

```python
import contextvars
import uuid
from functools import wraps

request_id_var: contextvars.ContextVar[str] = contextvars.ContextVar("request_id", default="-")

def new_request_id() -> str:
    return uuid.uuid4().hex[:12]

def set_request_id(rid: str | None = None) -> str:
    rid = rid or new_request_id()
    request_id_var.set(rid)
    return rid

def get_request_id() -> str:
    return request_id_var.get()

def with_request_id(func):
    @wraps(func)
    async def wrapper(*args, **kwargs):
        token = request_id_var.set(get_request_id())
        try:
            return await func(*args, **kwargs)
        finally:
            request_id_var.reset(token)
    return wrapper

# Usage at the API entry point
@app.post("/agent/run")
async def run_agent(body: RunRequest, request: Request):
    rid = request.headers.get("x-request-id") or new_request_id()
    set_request_id(rid)
    return await handle_agent_run(body)

# Every log line includes the ID automatically
logger.info("tool_call", extra={"request_id": get_request_id(), "tool": "web_search"})
```

- Contextvars preserve the ID across `await` points without threading it through every function signature
- Generate the ID once at the edge (or accept a client supplied one) and reset it on response so the context stays clean for the next request
- Works the same in TypeScript with `AsyncLocalStorage` so your frontend, gateway, and worker logs all share the same correlation key
