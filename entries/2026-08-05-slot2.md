Deterministic agent replay lets you re-run a failed session with the exact same LLM outputs to debug without burning tokens or hitting rate limits.

```python
import json
import os
from dataclasses import dataclass
from pathlib import Path
from typing import Any

@dataclass
class ReplayLog:
    session_id: str
    model: str
    prompt: str
    response: str
    tool_calls: list[dict[str, Any]]
    timestamp: float

class DeterministicReplay:
    def __init__(self, log_dir: str = ".replay_logs"):
        self.log_dir = Path(log_dir)
        self.log_dir.mkdir(exist_ok=True)

    def record(self, session_id: str, model: str, prompt: str, response: str, tool_calls: list[dict[str, Any]]) -> None:
        log = ReplayLog(
            session_id=session_id,
            model=model,
            prompt=prompt,
            response=response,
            tool_calls=tool_calls,
            timestamp=os.times().elapsed,
        )
        path = self.log_dir / f"{session_id}.jsonl"
        with path.open("a") as f:
            f.write(json.dumps(log.__dict__) + "\n")

    def replay(self, session_id: str) -> list[ReplayLog]:
        path = self.log_dir / f"{session_id}.jsonl"
        if not path.exists():
            return []
        logs = []
        with path.open() as f:
            for line in f:
                data = json.loads(line)
                logs.append(ReplayLog(**data))
        return logs

    def replay_response(self, session_id: str, step: int) -> str:
        """Return the exact LLM response for a given step without calling the model."""
        logs = self.replay(session_id)
        if step < len(logs):
            return logs[step].response
        raise IndexError("Step not found in replay log")
```

- Capture every LLM call with its prompt response and tool calls in a session-scoped log file
- Replay returns the recorded response for any step so you can iterate on downstream logic without re-prompting the model
- Store logs alongside code for reproducibility; a CI job can re-run the agent against golden logs to catch regressions
