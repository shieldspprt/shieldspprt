Stream partial tool-call results to the client so the UI renders incrementally instead of blocking on the slowest call.

```python
from typing import AsyncGenerator
from dataclasses import dataclass
import asyncio

@dataclass
class ToolEvent:
    tool_name: str
    status: str  # "start" | "delta" | "done" | "error"
    payload: dict | None = None
    error: str | None = None

async def stream_tool_results(
    tool_calls: list[dict],
    execute_fn
) -> AsyncGenerator[ToolEvent, None]:
    semaphore = asyncio.Semaphore(4)

    async def run_one(call: dict) -> AsyncGenerator[ToolEvent, None]:
        name = call["name"]
        args = call["arguments"]
        yield ToolEvent(name, "start", {"call_id": call["id"]})
        try:
            async with semaphore:
                async for chunk in execute_fn(name, args):
                    yield ToolEvent(name, "delta", chunk)
            yield ToolEvent(name, "done", {"call_id": call["id"]})
        except Exception as e:
            yield ToolEvent(name, "error", error=str(e))

    async def merge():
        async with asyncio.TaskGroup() as tg:
            for call in tool_calls:
                tg.create_task(
                    _drain(run_one(call))
                )

    async def _drain(gen):
        async for evt in gen:
            yield evt

    async for evt in merge():
        yield evt
```

- Bound concurrency with a semaphore prevents downstream API throttling
- The client renders each tool as it finishes rather than waiting for all
- Errors surface immediately per tool so the UI can show partial success
