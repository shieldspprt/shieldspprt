Stream LLM responses with AbortSignal so the client can cancel in-flight generations without leaking server resources. Pass the signal through the provider SDK and clean up the read stream on abort.

```typescript
import { streamText } from "ai";

export async function POST(req: Request) {
  const { messages } = await req.json();
  const abortSignal = req.signal;

  const result = streamText({
    model: "gpt-4o-mini",
    messages,
    abortSignal,
    onFinish: async ({ usage }) => {
      await logUsage(usage);
    },
  });

  return result.toDataStreamResponse({
    sendUsage: true,
    sendReasoning: true,
  });
}

// Client side: abort on navigation or new request
const controller = new AbortController();
const response = await fetch("/api/chat", {
  method: "POST",
  body: JSON.stringify({ messages }),
  signal: controller.signal,
});
// Call controller.abort() when the user sends a new message or navigates away
```

- Propagate `AbortSignal` through the provider SDK so upstream connections close promptly
- Use `onFinish` for usage logging only after the stream completes or aborts cleanly
- Client-side abort prevents wasted tokens and frees server concurrency slots for other requests
